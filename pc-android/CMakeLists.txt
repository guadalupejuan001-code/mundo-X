cmake_minimum_required(VERSION 3.16)
project(cosmic_pc LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
foreach(name x86_cpu boot_media pc_machine pc_iso)
    add_executable(${name}_test native/tests/${name}_test.cpp)
    target_compile_options(${name}_test PRIVATE -Wall -Wextra -Wpedantic)
    add_test(NAME ${name} COMMAND ${name}_test)
endforeach()
add_executable(cosmic_pc_cli native/pc_cli.cpp)
