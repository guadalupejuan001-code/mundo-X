cmake_minimum_required(VERSION 3.16)
project(mundo_x_native LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_executable(mundo_x native/src/main.cpp)
target_include_directories(mundo_x PRIVATE native/include)

add_executable(mundo_x_vm native/src/basis_vm_main.cpp)
target_include_directories(mundo_x_vm PRIVATE native/include)
add_executable(mundo_x_vm_test native/tests/basis_vm_test.cpp)
target_include_directories(mundo_x_vm_test PRIVATE native/include)
enable_testing()
add_test(NAME basis_vm COMMAND mundo_x_vm_test)

add_executable(mundo_x_sparse_observable_test native/tests/sparse_observable_test.cpp)
target_include_directories(mundo_x_sparse_observable_test PRIVATE native/include)
add_executable(mundo_x_quantum_echo_test native/tests/quantum_echo_test.cpp)
target_include_directories(mundo_x_quantum_echo_test PRIVATE native/include)
add_test(NAME sparse_observables COMMAND mundo_x_sparse_observable_test)
add_test(NAME quantum_echo COMMAND mundo_x_quantum_echo_test)

if (CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(mundo_x PRIVATE -O3 -Wall -Wextra -Wpedantic)
  target_compile_options(mundo_x_vm PRIVATE -O3 -Wall -Wextra -Wpedantic)
  target_compile_options(mundo_x_vm_test PRIVATE -O3 -Wall -Wextra -Wpedantic)
  target_compile_options(mundo_x_sparse_observable_test PRIVATE -O3 -Wall -Wextra -Wpedantic)
  target_compile_options(mundo_x_quantum_echo_test PRIVATE -O3 -Wall -Wextra -Wpedantic)
endif()
