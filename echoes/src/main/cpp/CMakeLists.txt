cmake_minimum_required(VERSION 3.22.1)
project(mundox_echoes LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(MUNDO_X_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
add_library(mundox_echoes SHARED native-echo.cpp)
target_include_directories(mundox_echoes PRIVATE "${MUNDO_X_ROOT}/native/include")
target_compile_options(mundox_echoes PRIVATE -O3 -Wall -Wextra -Wpedantic)
target_link_options(mundox_echoes PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
